```javascript
      metrics.bugPreventionValue += 50000; // Average cost of production bug
    }
  });
  
  const totalValue = Object.values(metrics).reduce((sum, value) => sum + value, 0);
  const roi = (totalValue - (timeInvested * 200)) / (timeInvested * 200) * 100; // $200/hour rate
  
  return { metrics, totalValue, roi };
};
```
